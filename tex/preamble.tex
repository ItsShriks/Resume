%-----------RESUME COMMANDS-----------------
% Entry with a one-line header (jobs, degrees, projects):
%   \resumeSubheading{Bold title}{right: dates / link}{after the dash: organisation / degree / tech}{note in brackets: place / "Master Thesis"}
% renders as   **Title** — Organisation (Note)                    Dates
\newcommand{\resumeSubheading}[4]{%
  \item[]\hspace*{-1.1em}%   flush with the section title
  \begin{tabularx}{\dimexpr\linewidth+1.1em}{@{}>{\raggedright\arraybackslash}X@{\hspace{8pt}}r@{}}
    \textbf{#1}\if\relax\detokenize{#3}\relax\else\ \textbf{---}\ #3\fi
    \if\relax\detokenize{#4}\relax\else\ {\color{subtle}(#4)}\fi
    & #2 \\
  \end{tabularx}%
}
